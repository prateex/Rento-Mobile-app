// WORKFLOW
// ============================================================================
//
// Step 1: CREATE AUTH USER
//   Input: email, password
//   Output: auth.user.id (copy this)
//   Uses: Supabase Admin API (/auth/v1/admin/users)
//
// Step 2: CREATE SHOP
//   Input: name, address, city, state, phone, email, gst
//   Output: shop.id (copy this)
//   SQL: INSERT INTO rental_shops
//
// Step 3: ASSIGN OWNER
//   Input: auth.user.id, shop.id, name, phone, email
//   Role: Fixed to 'owner' (cannot change)
//   SQL: INSERT INTO users with role='owner'
//
// Step 4: (OPTIONAL) ADD STAFF
//   Input: auth.user.id, shop.id, name, phone, email, role
//   Role: Can be 'staff' or 'admin'
//   SQL: INSERT INTO users with explicit role
//
// Step 5: VALIDATE
//   Queries: users, shops, auth context
//   Verifies: role is explicit, shop_id present, RLS not blocking
//
// ============================================================================
// SCHEMA NOTES (DO NOT INVENT)
// ============================================================================
//
// users table:
//   ✓ id (UUID, PK)
//   ✓ auth_id (UUID, UNIQUE, NOT NULL) — links to auth.users
//   ✓ shop_id (UUID, NOT NULL, FK) — links to rental_shops
//   ✓ name (TEXT, NOT NULL)
//   ✓ phone (TEXT)
//   ✓ email (TEXT)
//   ✓ role (user_role enum, NOT NULL) — EXPLICIT, NO DEFAULT
//   ✓ is_active (BOOLEAN, DEFAULT true)
//
// rental_shops table:
//   ✓ id (UUID, PK)
//   ✓ owner_id (UUID, NOT NULL, FK) — links to auth.users
//   ✓ name (TEXT, NOT NULL)
//   ✓ address (TEXT)
//   ✓ city (TEXT)
//   ✓ state (TEXT)
//   ✓ pincode (TEXT)
//   ✓ phone (TEXT)
//   ✓ email (TEXT)
//   ✓ gst_number (TEXT)
//
// user_role enum values:
//   - 'owner'
//   - 'staff'
//   - 'admin' (for this page, but typically 'owner' for setup)
//
// ============================================================================
// RLS RULES (WHY THIS PAGE MATTERS)
// ============================================================================
//
// OLD (BROKEN):
//   CREATE POLICY shop_access ON users
//   USING (shop_id IN (SELECT shop_id FROM users ...))
//   → Recursion, 500 error, login loop
//
// NEW (FIXED):
//   CREATE POLICY view_own_record ON users FOR SELECT
//   USING (auth_id = auth.uid())
//   → Direct check, no recursion, login works
//
// This page bypasses RLS during setup (service role), then the app 
// respects RLS during normal operation (anon key).
//
// ============================================================================
// ERROR MESSAGES & FIXES
// ============================================================================
//
// "VITE_SUPABASE_SERVICE_ROLE_KEY not set"
//   → Add to .env.local, restart dev server
//
// "Auth API error: ..."
//   → Check VITE_SUPABASE_URL and service role key
//
// "User already exists"
//   → Use different email, or manually delete from auth
//
// "Shop name required"
//   → Fill in all required fields
//
// "Auth user ID required"
//   → Complete Step 1 first
//
// "Duplicate user record"
//   → auth_id already has a public.users row
//   → Choose different auth user
//
